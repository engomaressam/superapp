Agent Base Class

class BaseAgent: """Base class for all ARIA agents""" name: str = "BaseAgent" description: str = "" async def can_handle(self, task: Task) -> bool: """Check if agent can handle this task""" raise NotImplementedError async def plan(self, task: Task, context: dict) -> List[dict]: """Create execution plan for the task""" raise NotImplementedError async def execute(self, task: Task, context: dict) -> AgentResult: """Execute the task and return result""" plan = await self.plan(task, context) for step in plan: result = await self._execute_step(step) return result
